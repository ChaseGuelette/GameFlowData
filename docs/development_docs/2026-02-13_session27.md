**Changes:**

```python
from concurrent.futures import ThreadPoolExecutor, as_completed

def _build_features_df(self, players: list[dict], target_date: date) -> pd.DataFrame:
    start_time = time.perf_counter()

    def fetch_single(player: dict) -> pd.DataFrame | None:
        try:
            features = self.feature_store.get_player_game_features(...)
            if features is not None:
                row_df = pd.DataFrame([features]) if isinstance(features, dict) else features
                row_df["player_id"] = player["player_id"]
                return row_df
        except Exception as e:
            logger.error(f"Error building features for player {player['player_id']}: {e}")
        return None

    all_features = []
    max_workers = min(8, len(players)) if players else 1

    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        futures = {executor.submit(fetch_single, p): p for p in players}
        for future in as_completed(futures):
            result = future.result()
            if result is not None:
                all_features.append(result)

    elapsed = time.perf_counter() - start_time
    logger.info(f"Built features for {len(all_features)} players in {elapsed:.1f}s ({max_workers} workers)")
    ...
```

### 2. Prop Lines Query Optimization (685x faster)

**Before:** 137 seconds
- `LPAD()` function in WHERE clause prevented index usage
- Full table scan on 26.2M rows

**After:** 0.2 seconds (685x faster)
- Query searches both 8-digit and 10-digit game_id formats
- `LPAD()` removed from WHERE and PARTITION BY clauses
- Indexes created on game_id column

**Query Change:**
```python
# Create both 8-digit and 10-digit game_ids to search
game_ids_10digit = [g.zfill(10) for g in game_ids]
game_ids_8digit = [g.lstrip('0') for g in game_ids]
all_game_ids = list(set(game_ids_10digit + game_ids_8digit))

query = text("""
    WITH ranked_lines AS (
        SELECT player_id, game_id, bookmaker, market_key, line, outcome_label, odds_american, snapshot_time,
            ROW_NUMBER() OVER (
                PARTITION BY player_id, game_id, market_key, bookmaker, line, outcome_label
                ORDER BY snapshot_time DESC
            ) as rn
        FROM raw_player_props_combined
        WHERE game_id IN :game_ids  -- No LPAD here!
          AND market_key IN :markets
          AND player_id IS NOT NULL
    )
    SELECT player_id, LPAD(game_id, 10, '0') as game_id, ...
""")
```

**Indexes Created (via Supabase Dashboard):**
- `idx_props_game_id` on `raw_player_props_combined(game_id)`
- `idx_props_game_market` on `raw_player_props_combined(game_id, market_key)`
- `idx_props_game_id_padded` on `raw_player_props_combined(LPAD(game_id, 10, '0'))`
