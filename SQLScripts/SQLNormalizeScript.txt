-- First, let's make a backup of the table (always recommended before mass updates)
-- Create a backup first
CREATE TABLE "all_player_game_stats_2024-25_backup" AS 
SELECT * FROM "all_player_game_stats_2024-25";

-- Update the dates to a consistent YYYY-MM-DD format
UPDATE "advanced_player_clutch_stats_by_date"
SET "DATE_TO" = 
    CASE 
        -- When it's already in YYYY-MM-DD format
        WHEN "DATE_TO" ~ '^\d{4}-\d{2}-\d{2}$' THEN "DATE_TO"
        
        -- When it has timestamp component (YYYY-MM-DD HH:MI:SS)
        WHEN "DATE_TO" ~ '^\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}$' THEN 
            SUBSTRING("DATE_TO" FROM 1 FOR 10)
            
        -- When it's in MM/DD/YYYY format (like "12/28/2020")
        WHEN "DATE_TO" ~ '^\d{1,2}/\d{1,2}/\d{4}$' THEN 
            TO_CHAR(TO_DATE("DATE_TO", 'MM/DD/YYYY'), 'YYYY-MM-DD')
            
        -- When it's in "MON DD, YYYY" format (like "JAN 09, 2025")
        WHEN "DATE_TO" ~ '^[A-Za-z]{3} \d{1,2}, \d{4}$' THEN 
            TO_CHAR(TO_DATE("DATE_TO", 'MON DD, YYYY'), 'YYYY-MM-DD')
            
        -- Add more patterns if needed
        
        ELSE "DATE_TO" -- Keep as is if no patterns match
    END;
